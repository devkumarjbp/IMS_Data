पेज लोड/रन होते ही ऑटो-फ़ेच (Auto-Sync): जैसे ही आप ऐप खोलेंगे या लॉगिन करेंगे, यह बैकग्राउंड में Google Sheet से सारा ताज़ा डेटा लोड करके टेबल में भर देगा।

स्मार्ट स्टेटस इंडिकेटर: हेडर में एक लाइव इंडिकेटर दिखेगा—"🟢 सिंक हो गया", "🟡 सिंक हो रहा है...", या "🔴 ऑफलाइन"।

शीट में डेटा पढ़ने के लिए doGet सपोर्ट: Google Apps Script में doGet जोड़ दिया गया है ताकि वेब ऐप शीट का सारा डेटा JSON फ़ॉर्मैट में सीधे ब्राउज़र को भेज सके।

// --- Google Apps Script (Auto-Sync Read + Write) ---
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  if (rows.length < 2) {
    return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  }
  
  var headers = rows[0];
  var data = [];
  for (var i = 1; i < rows.length; i++) {
    var row = rows[i];
    var item = {};
    for (var j = 0; j < headers.length; j++) {
      item[headers[j]] = row[j];
    }
    data.push(item);
  }
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["ID", "पूरा नाम", "मोबाइल नंबर", "ईमेल पता", "श्रेणी / स्टेटस", "तारीख", "टिप्पणी / विवरण"]);
    }
    
    var row = [
      data.id || "",
      data.fullName || "",
      data.phone || "",
      data.email || "",
      data.category || "",
      data.entryDate || "",
      data.notes || ""
    ];
    sheet.appendRow(row);
    return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  if (rows.length < 2) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  
  var headers = rows[0];
  var data = [];
  for (var i = 1; i < rows.length; i++) {
    var row = rows[i];
    var item = {};
    for (var j = 0; j < headers.length; j++) {
      item[headers[j]] = row[j];
    }
    data.push(item);
  }
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);

    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "ID", "Name of IA", "Phase", "Plot No.", "Sector", 
        "Length", "Width", "Area in Sqmtr", "Area in Hect", 
        "Area in Acre", "Status of Plot"
      ]);
    }

    var row = [
      data.id || ("PLOT-" + Date.now().toString().slice(-5)),
      data.nameOfIA || "",
      data.phase || "",
      data.plotNo || "",
      data.sector || "",
      data.length || "",
      data.width || "",
      data.areaSqmtr || "",
      data.areaHect || "",
      data.areaAcre || "",
      data.statusOfPlot || ""
    ];
    sheet.appendRow(row);
    return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

// --- Apps Script for Multi-Sheet CRUD (Sheet1 & PlotMaster) ---

function doGet(e) {
  var sheetName = (e && e.parameter && e.parameter.sheet) ? e.parameter.sheet : "Sheet1";
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(sheetName);
  
  if (!sheet) {
    return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  }
  
  var rows = sheet.getDataRange().getValues();
  if (rows.length < 2) {
    return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  }
  
  var headers = rows[0];
  var data = [];
  for (var i = 1; i < rows.length; i++) {
    var row = rows[i];
    var item = {};
    for (var j = 0; j < headers.length; j++) {
      item[headers[j]] = row[j];
    }
    data.push(item);
  }
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var payload = JSON.parse(e.postData.contents);
    var sheetName = payload.targetSheet || "PlotMaster";
    var action = payload.action || "create";
    var data = payload.data || payload;
    
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
    }
    
    // Default Headers for PlotMaster
    if (sheetName === "PlotMaster" && sheet.getLastRow() === 0) {
      sheet.appendRow(["ID", "Name of IA", "Phase", "Plot No.", "Sector", "Length", "Width", "Area in Sqmtr", "Area in Hect", "Area in Acre", "Status of Plot"]);
    }
    
    var idColumnIndex = 1; // Column A (ID)
    var allData = sheet.getDataRange().getValues();
    
    if (action === "create") {
      var row = [
        data.id || ("PLOT-" + Date.now().toString().slice(-5)),
        data.nameOfIA || "",
        data.phase || "",
        data.plotNo || "",
        data.sector || "",
        data.length || "",
        data.width || "",
        data.areaSqmtr || "",
        data.areaHect || "",
        data.areaAcre || "",
        data.statusOfPlot || ""
      ];
      sheet.appendRow(row);
      return ContentService.createTextOutput(JSON.stringify({ status: "success", action: "create" })).setMimeType(ContentService.MimeType.JSON);
    }
    
    if (action === "update") {
      var targetId = data.id;
      for (var r = 1; r < allData.length; r++) {
        if (String(allData[r][0]) === String(targetId)) {
          var updatedRow = [
            targetId,
            data.nameOfIA || "",
            data.phase || "",
            data.plotNo || "",
            data.sector || "",
            data.length || "",
            data.width || "",
            data.areaSqmtr || "",
            data.areaHect || "",
            data.areaAcre || "",
            data.statusOfPlot || ""
          ];
          sheet.getRange(r + 1, 1, 1, updatedRow.length).setValues([updatedRow]);
          return ContentService.createTextOutput(JSON.stringify({ status: "success", action: "update" })).setMimeType(ContentService.MimeType.JSON);
        }
      }
      // If not found, append
      sheet.appendRow([targetId, data.nameOfIA, data.phase, data.plotNo, data.sector, data.length, data.width, data.areaSqmtr, data.areaHect, data.areaAcre, data.statusOfPlot]);
      return ContentService.createTextOutput(JSON.stringify({ status: "success", action: "created_on_update" })).setMimeType(ContentService.MimeType.JSON);
    }
    
    if (action === "delete") {
      var deleteId = data.id;
      for (var r = 1; r < allData.length; r++) {
        if (String(allData[r][0]) === String(deleteId)) {
          sheet.deleteRow(r + 1);
          return ContentService.createTextOutput(JSON.stringify({ status: "success", action: "delete" })).setMimeType(ContentService.MimeType.JSON);
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "not_found" })).setMimeType(ContentService.MimeType.JSON);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "invalid_action" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
